import "server-only";

import { cookies } from "next/headers";

/**
 * Nombre de la cookie que almacena
 * temporalmente el JWT administrativo.
 */
const ADMIN_SESSION_COOKIE = "admin_session";

/**
 * Duración máxima permitida:
 * 15 minutos, igual que el JWT de Spring Boot.
 */
const MAX_SESSION_SECONDS = 15 * 60;

/**
 * Guarda el JWT en una cookie HttpOnly.
 *
 * Solo debe llamarse después de que
 * Spring Boot autentique al administrador.
 *
 * Debe ejecutarse dentro de una Server Action
 * o un Route Handler de Next.js.
 */
export async function createAdminSession(
  accessToken: string,
  expiresIn: number
): Promise<void> {
  if (
    typeof accessToken !== "string" ||
    accessToken.trim().length === 0
  ) {
    throw new Error("Token de sesión inválido.");
  }

  if (
    !Number.isSafeInteger(expiresIn) ||
    expiresIn <= 0
  ) {
    throw new Error("Duración de sesión inválida.");
  }

  const sessionDuration = Math.min(
    expiresIn,
    MAX_SESSION_SECONDS
  );

  const cookieStore = await cookies();

  cookieStore.set(
    ADMIN_SESSION_COOKIE,
    accessToken,
    {
      httpOnly: true,
      secure: process.env.NODE_ENV === "production",
      sameSite: "lax",
      path: "/",
      maxAge: sessionDuration,
      priority: "high",
    }
  );
}

/**
 * Recupera el JWT de la sesión actual.
 *
 * IMPORTANTE:
 * Leer una cookie NO significa que el JWT
 * sea válido ni que el usuario esté autorizado.
 *
 * Spring Boot deberá validar el JWT antes
 * de permitir cualquier operación administrativa.
 */
export async function getAdminSessionToken():
  Promise<string | null> {
  const cookieStore = await cookies();

  const token = cookieStore.get(
    ADMIN_SESSION_COOKIE
  )?.value;

  if (!token || token.trim().length === 0) {
    return null;
  }

  return token;
}

/**
 * Elimina la cookie de sesión.
 *
 * Debe ejecutarse desde una Server Action
 * o un Route Handler de Next.js.
 */
export async function deleteAdminSession():
  Promise<void> {
  const cookieStore = await cookies();

  cookieStore.delete(ADMIN_SESSION_COOKIE);
}

import "server-only";

import { redirect } from "next/navigation";

import {
  getAdminSessionToken,
} from "@/lib/auth/admin-session";

const API_BASE_URL = (
  process.env.API_BASE_URL ??
  "http://127.0.0.1:8080/api/v1"
).replace(/\/+$/, "");

/**
 * Redirige al login cuando Spring Boot
 * rechaza la autenticación o autorización.
 *
 * Puede reutilizarse desde clientes HTTP
 * ejecutados exclusivamente en el servidor.
 */
export function redirectIfAdminUnauthorized(
  status: number
): void {
  if (status === 401 || status === 403) {
    redirect("/admin/login");
  }
}

/**
 * Verifica que el administrador tenga
 * una sesión válida y autorizada.
 *
 * Si falta la cookie o Spring Boot rechaza
 * el JWT, redirige a /admin/login.
 *
 * Los errores de conexión o los errores 5xx
 * no se confunden con sesiones vencidas.
 */
export async function requireAdminSession():
  Promise<void> {

  // Recuperar JWT desde la cookie HttpOnly.
  const token = await getAdminSessionToken();

  // No existe una sesión administrativa.
  if (!token) {
    redirect("/admin/login");
  }

  // Consultar un endpoint protegido.
  const response = await fetch(
    `${API_BASE_URL}/admin/products`,
    {
      method: "GET",
      cache: "no-store",
      headers: {
        Accept: "application/json",
        Authorization: `Bearer ${token}`,
      },
      signal: AbortSignal.timeout(10000),
    }
  );

  // Rechazar JWT inválido, vencido o sin permisos.
  redirectIfAdminUnauthorized(response.status);

  // No ocultar problemas reales del backend.
  if (!response.ok) {
    throw new Error(
      "No fue posible verificar la sesión administrativa."
    );
  }
}

import {
  getAdminMutationToken,
} from "@/lib/auth/admin-mutation-token";

const API_BASE_URL =
  process.env.API_BASE_URL ??
  "http://127.0.0.1:8080/api/v1";

const UUID_REGEX =
  /^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/i;

export type ProductStatusResult =
  | { success: true }
  | { success: false; message: string };

type StatusField = "active" | "available";

interface ProductStatusResponse {
  id: string;
  active: boolean;
  available: boolean;
}

/**
 * Función interna para actualizar un estado.
 *
 * No se exporta como Server Action.
 *
 * Exige JWT antes de realizar cualquier
 * modificación en Spring Boot.
 */
async function updateStatus(
  productId: string,
  field: StatusField,
  value: boolean
): Promise<ProductStatusResult> {

  // Mantener la restricción actual
  // del backend de desarrollo local.
  if (process.env.NODE_ENV !== "development") {
    return {
      success: false,
      message:
        "La administración solo está habilitada en desarrollo local.",
    };
  }

  // Recuperar la sesión del administrador.
  const token = await getAdminMutationToken();

  // No permitir modificaciones sin sesión.
  if (!token) {
    return {
      success: false,
      message:
        "Tu sesión ha vencido o no has iniciado sesión. Ingresa nuevamente.",
    };
  }

  // Validar identificador.
  if (
    typeof productId !== "string" ||
    !UUID_REGEX.test(productId)
  ) {
    return {
      success: false,
      message: "El identificador del producto no es válido.",
    };
  }

  // Validar estado.
  if (typeof value !== "boolean") {
    return {
      success: false,
      message: "El estado enviado no es válido.",
    };
  }

  const endpoint =
    field === "active"
      ? "publication"
      : "availability";

  try {
    // Enviar la modificación autenticada
    // al backend Spring Boot.
    const response = await fetch(
      `${API_BASE_URL}/admin/products/${productId}/${endpoint}`,
      {
        method: "PATCH",
        cache: "no-store",
        headers: {
          "Content-Type": "application/json",
          Accept: "application/json",
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({
          [field]: value,
        }),
      }
    );

    // Sesión vencida o token rechazado.
    if (
      response.status === 401 ||
      response.status === 403
    ) {
      return {
        success: false,
        message:
          "Tu sesión no está autorizada. Inicia sesión nuevamente.",
      };
    }

    // Producto inexistente.
    if (response.status === 404) {
      return {
        success: false,
        message: "El producto ya no existe.",
      };
    }

    // Solicitud inválida.
    if (response.status === 400) {
      return {
        success: false,
        message:
          "Spring Boot rechazó el cambio solicitado.",
      };
    }

    // Conflicto con las reglas del backend.
    if (response.status === 409) {
      return {
        success: false,
        message:
          "No se puede realizar el cambio debido a una restricción del backend.",
      };
    }

    if (!response.ok) {
      console.error(
        "Error actualizando estado:",
        response.status
      );

      return {
        success: false,
        message:
          "No fue posible actualizar el estado del producto.",
      };
    }

    // Confirmar que el backend devolvió
    // el estado solicitado.
    const updated: ProductStatusResponse =
      await response.json();

    if (
      updated.id.toLowerCase() !==
        productId.toLowerCase() ||
      updated[field] !== value
    ) {
      return {
        success: false,
        message:
          "No se pudo confirmar el estado actualizado. Recarga la página antes de reintentar.",
      };
    }

    return { success: true };

  } catch (error) {
    console.error(
      "Error de conexión con Spring Boot:",
      error
    );

    return {
      success: false,
      message:
        "No se pudo completar la operación. Comprueba el estado del producto antes de reintentar.",
    };
  }
}

export async function updateProductPublicationAction(

  return updateStatus(

 * Marcar un producto como disponible

 */
export async function updateProductAvailabilityAction(
  productId: string,

    available
  );
}

 */
export async function createProductAction(
  input: CreateProductInput

  // 1. Conservar la restricción temporal
  // del entorno de desarrollo local.
  if (process.env.NODE_ENV !== "development") {
    return {

  if (!token) {
    return {
      success: false,
      message:
        "Tu sesión ha vencido o no has iniciado sesión. Ingresa nuevamente.",
    };
  }

  // 3. Validar la estructura de los datos.
  if (
    !input ||
    typeof input.name !== "string" ||
    typeof input.description !== "string" ||
    typeof input.price !== "number" ||
    typeof input.categoryId !== "string"
  ) {
    return {
      success: false,
      message: "Los datos del producto no son válidos.",
    };

  const description = input.description.trim();
  const categoryId = input.categoryId;

  // Validar nombre.
  if (name.length < 2 || name.length > 150) {
    return {
      success: false,
      message:
        "El nombre debe tener entre 2 y 150 caracteres.",

    return {
      success: false,
      message:
        "La descripción no puede superar los 2000 caracteres.",

    !Number.isFinite(input.price) ||
    input.price < 0 ||
    !Number.isInteger(input.price) ||
    input.price > 9999999999
  ) {
    return {
      success: false,
      message:
        "Ingresa un precio válido en pesos colombianos.",
    };
  }

  // Validar identificador de categoría.
  if (!UUID_REGEX.test(categoryId)) {
    return {
      success: false,
      message:
        "Selecciona una categoría válida.",
    };
  }

  try {

    // 4. Enviar el producto a Spring Boot
    // utilizando la sesión administrativa.
    const response = await fetch(
      `${API_BASE_URL}/admin/products`,
      {
        method: "POST",
        cache: "no-store",

    if (response.status === 404) {
      return {
        success: false,
        message:
          "La categoría seleccionada no existe.",
      };
    }

    // Datos rechazados por Spring Boot.
    if (response.status === 400) {
      return {
        success: false,
        message:
          "Spring Boot rechazó los datos del producto.",
      };
    }

    // Conflicto con reglas de negocio.
    if (response.status === 409) {
      return {
        success: false,
        message:
          "No fue posible crear el producto debido a una restricción del backend.",
      };
    }

    // Otros errores HTTP.
    if (!response.ok) {
      console.error(
        "Error creando producto:",
        response.status
      );

      return {
        success: false,
        message:
          "No fue posible guardar el producto.",
      };
    }

    // 5. Obtener el UUID real generado
    // por Spring Boot.
    const product: unknown = await response.json();

    if (
      typeof product !== "object" ||
      product === null ||
      !("id" in product) ||
      typeof product.id !== "string" ||
      !UUID_REGEX.test(product.id)
    ) {
      return {
        success: false,
        message:
          "El servidor respondió correctamente, pero no se pudo confirmar el identificador del producto. Consulta el listado antes de reintentar.",
      };
    }

    // Producto guardado y confirmado.
    return {
      success: true,
      productId: product.id,
    };

  } catch (error) {
    console.error(
      "Error de conexión al crear producto:",
      error
    );

    return {
      success: false,
      message:

const ALLOWED_TYPES = [

  | {

    }
  | {
      success: false;

/**
 * Valida una respuesta de imagen de Spring Boot.
 */
function isImageResponse(
  value: unknown
): value is {
  id: string;
  imageUrl: string;
} {
  if (
    typeof value !== "object" ||
    value === null
  ) {
    return false;
  }

  if (
    !("id" in value) ||
    !("imageUrl" in value)
  ) {
    return false;
  }

  return (
    typeof value.id === "string" &&
    UUID_REGEX.test(value.id) &&
    typeof value.imageUrl === "string" &&
    value.imageUrl.startsWith("https://")
  );
}

 *
 * Flujo:
 * Next.js -> Spring Boot -> Cloudinary
 *                         -> PostgreSQL
 *
 * La solicitud exige JWT administrativo.

): Promise<UploadProductImageResult> {

  // del entorno de desarrollo.

      message:

    };

  if (!token) {
    return {
      success: false,
      message: SESSION_REQUIRED_MESSAGE,
    };
  }

  // 3. Validar UUID del producto.
  if (
    typeof productId !== "string" ||
    !UUID_REGEX.test(productId)
  ) {
    return {
      success: false,
      message:
        "El identificador del producto no es válido.",
    };
  }

    return {
      success: false,
      message:
        "Debes seleccionar una fotografía válida.",
    };
  }

  // Recuperar el campo esperado por Spring Boot.
  const value = formData.get("file");

  if (!(value instanceof File)) {
    return {
      success: false,
      message:
        "Debes seleccionar una fotografía válida.",
    };
  }

  // No permitir archivos vacíos.
  if (value.size === 0) {
    return {
      success: false,
      message:
        "No se permiten archivos vacíos.",
    };
  }

  // Validar JPG o PNG.
  if (!ALLOWED_TYPES.includes(value.type)) {
    return {
      success: false,
      message:
        "Solo se permiten fotografías JPG o PNG.",
    };
  }

  // Validar tamaño máximo de 5 MiB.
  if (value.size > MAX_FILE_SIZE) {
    return {
      success: false,
      message:
        "La fotografía supera el límite de 5 MiB.",
    };
  }

  try {

    // 5. Preparar multipart/form-data.
    const uploadData = new FormData();

    uploadData.append(
      "file",
      value,
      value.name
    );

    // 6. Enviar fotografía a Spring Boot
    // con autorización JWT.
    const response = await fetch(
      `${API_BASE_URL}/admin/products/${productId}/images/upload`,
      {
        method: "POST",
        cache: "no-store",
        headers: {
          Accept: "application/json",
          Authorization: `Bearer ${token}`,
        },
        body: uploadData,
      }
    );

    // No agregar Content-Type manualmente:
    // fetch genera el boundary de multipart.

    // JWT inválido, vencido o sin permisos.
    if (
      response.status === 401 ||
      response.status === 403
    ) {
      return {
        success: false,
        message: SESSION_UNAUTHORIZED_MESSAGE,
      };
    }

    // Producto inexistente.
    if (response.status === 404) {
      return {
        success: false,
        message:
          "El producto no existe en PostgreSQL.",
      };
    }

    // Archivo inválido.
    if (response.status === 400) {
      return {
        success: false,
        message:
          "Spring Boot rechazó la fotografía. Comprueba que sea JPG o PNG.",
      };
    }

    // Límite de cinco fotografías.
    if (response.status === 409) {
      return {
        success: false,
        message:
          "El producto ya tiene el máximo de cinco fotografías.",
      };
    }

    // Límite de tamaño.
    if (response.status === 413) {
      return {
        success: false,
        message:
          "La fotografía supera el tamaño permitido por el servidor.",
      };
    }

    if (!response.ok) {
      console.error(
        "Error HTTP al subir fotografía:",
        response.status
      );

      return {
        success: false,
        message:
          "No fue posible guardar la fotografía.",
      };
    }

    // 7. Confirmar la respuesta de Spring Boot.
    const image: unknown = await response.json();

    if (!isImageResponse(image)) {
      return {
        success: false,
        message:
          "El servidor aceptó la fotografía, pero no se pudo confirmar su información. Revisa las imágenes del producto antes de intentar subirla nuevamente.",
      };
    }

    return {
      success: true,
      imageId: image.id,
      imageUrl: image.imageUrl,
    };

  } catch (error) {
    console.error(
      "Error enviando fotografía a Spring Boot:",
      error
    );

    return {
      success: false,
      message:
        "No se pudo completar la carga. Verifica las fotografías existentes antes de reintentar.",
    };
  }
}

      updatedCategory.active !== active
    ) {
      return {
        success: false,
        message:
          "El backend no confirmó el estado solicitado. " +
          "Actualiza la página y verifica la categoría.",
      };
    }

    return {
      success: true,
    };

  } catch (error) {
    console.error(
      "Error de conexión al modificar categoría:",
      error
    );

    return {
      success: false,
      message:
        "No se pudo conectar con Spring Boot. " +
        "Verifica que esté ejecutándose.",
    };
  }
}
